cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(FileTraceStudio VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Use Windows x64 with the Visual Studio C++ tools.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(ENABLE_ASAN "Enable AddressSanitizer for offline tests" OFF)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc)
if(ENABLE_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/INCREMENTAL:NO /DEBUG:FULL)
endif()
add_library(FileTrace STATIC SDK/FileTrace.cpp App/Model.cpp)
target_include_directories(FileTrace PUBLIC SDK App)
target_link_libraries(FileTrace PUBLIC fltlib user32)
add_executable(FileTraceStudio WIN32 App/Main.cpp)
target_link_libraries(FileTraceStudio PRIVATE FileTrace gdiplus gdi32 comdlg32 shell32)
target_link_options(FileTraceStudio PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'" "/PDBALTPATH:%_PDB%")
add_executable(FileTraceCLI App/Console.cpp)
target_link_libraries(FileTraceCLI PRIVATE FileTrace)
add_executable(FileTraceTests Tests/Tests.cpp)
target_link_libraries(FileTraceTests PRIVATE FileTrace)
add_executable(CaptureExample Examples/Capture.cpp)
target_link_libraries(CaptureExample PRIVATE FileTrace)
foreach(target FileTraceStudio FileTraceCLI FileTraceTests CaptureExample)
 set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
enable_testing()
add_test(NAME OfflineProtocolBoundsAndRundownModel COMMAND FileTraceTests)
add_test(NAME OfflineCliSample COMMAND FileTraceCLI --sample)
